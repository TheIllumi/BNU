class OptimizationWorker(QThread):
    """Background thread for running optimization scripts."""
    
    progress_update = pyqtSignal(str)
    progress_value = pyqtSignal(int)
    finished_signal = pyqtSignal(bool, str)
    
    def __init__(self, config_dict, scripts_to_run):
        super().__init__()
        self.config_dict = config_dict
        self.scripts_to_run = scripts_to_run
        
    def run(self):
        try:
            self.progress_update.emit("🚀 Starting optimization process...")
            self.progress_value.emit(10)
            
            # Save configuration file
            os.makedirs(DATA_DIR, exist_ok=True)
            config_path = os.path.join(DATA_DIR, CONFIG_FILE)
            with open(config_path, 'w') as f:
                json.dump(self.config_dict, f, indent=4)
            
            self.progress_update.emit("💾 Configuration saved successfully")
            self.progress_value.emit(20)
            
            # Run each script
            for i, script_path in enumerate(self.scripts_to_run, 1):
                script_name = os.path.basename(script_path)
                progress = 20 + (i * 60 // len(self.scripts_to_run))
                
                self.progress_update.emit(f"📋 Running {script_name}...")
                self.progress_value.emit(progress)
                
                if not os.path.exists(script_path):
                    self.finished_signal.emit(False, f"❌ Script not found: {script_path}")
                    return
                
                # Run the script with comprehensive error handling
                try:
                    result = subprocess.run(
                        [sys.executable, script_path], 
                        capture_output=True, 
                        text=True,
                        encoding='utf-8',
                        errors='replace',  # Replace invalid characters instead of crashing
                        cwd=os.path.dirname(os.path.abspath(script_path)) if os.path.dirname(script_path) else ".",
                        timeout=300  # 5 minute timeout
                    )
                except subprocess.TimeoutExpired:
                    self.finished_signal.emit(False, f"❌ {script_name} timed out after 5 minutes")
                    return
                except Exception as e:
                    self.finished_signal.emit(False, f"❌ Error running {script_name}: {str(e)}")
                    return
                
                if result.returncode != 0:
                    error_msg = f"❌ {script_name} failed:\n{result.stderr}\n{result.stdout}"
                    self.finished_signal.emit(False, error_msg)
                    return
                
                self.progress_update.emit(f"✅ {script_name} completed successfully")
            
            self.progress_value.emit(100)
            self.progress_update.emit("🎉 All optimization scripts completed successfully!")
            self.finished_signal.emit(True, "✅ Schedule generation completed successfully!")
            
        except Exception as e:
            self.finished_signal.emit(False, f"❌ Unexpected error: {str(e)}")

    def connect_signals(self):
        """Connect button signals to their respective functions."""
        self.add_btn.clicked.connect(self.add_row)
        self.remove_btn.clicked.connect(self.remove_row)
        self.undo_btn.clicked.connect(self.undo_remove)
        self.import_btn.clicked.connect(self.import_data)
        self.export_btn.clicked.connect(self.export_data)
        
    def load_data(self):
        """Load data from CSV file with error handling."""
        os.makedirs(DATA_DIR, exist_ok=True)
        
        if not os.path.exists(self.csv_file):
            return
            
        try:
            with open(self.csv_file, 'r', newline='', encoding='utf-8') as f:
                reader = csv.reader(f)
                data = list(reader)
                
            if not data:
                return
                
            # Skip header row
            data_rows = data[1:] if len(data) > 1 else []
            self.table.setRowCount(len(data_rows))
            
            for row_idx, row_data in enumerate(data_rows):
                for col_idx, cell_data in enumerate(row_data):
                    if col_idx < len(self.columns):
                        # Check if this column should have a dropdown
                        if col_idx in self.dropdown_columns:
                            combo = QComboBox()
                            combo.addItems(self.dropdown_columns[col_idx])
                            if cell_data in self.dropdown_columns[col_idx]:
                                combo.setCurrentText(cell_data)
                            combo.setStyleSheet("""
                                QComboBox {
                                    background-color: #3a3a3a;
                                    border-radius: 4px;
                                    min-height: 10px;
                                    color: white;
                                }
                            """)
                            self.table.setCellWidget(row_idx, col_idx, combo)
                        else:
                            item = QTableWidgetItem(cell_data)
                            # Highlight required columns
                            if col_idx in [i for i, col in enumerate(self.columns) if col in self.required_columns]:
                                item.setBackground(QColor(70, 130, 180, 50))
                            self.table.setItem(row_idx, col_idx, item)
                            
            self.table.horizontalHeader().setSectionResizeMode(QHeaderView.ResizeMode.Stretch)
            
        except Exception as e:
            QMessageBox.critical(self, "Error", f"Failed to load data from {self.csv_file}:\n{str(e)}")
            
    def add_row(self):
        """Add a new row with animated feedback."""
        row_count = self.table.rowCount()
        self.table.insertRow(row_count)
        
        # Add dropdown widgets for appropriate columns
        for col_idx, options in self.dropdown_columns.items():
            combo = QComboBox()
            combo.addItems(options)
            combo.setStyleSheet("""
                QComboBox {
                    background-color: #3a3a3a;
                    border: 1px solid #555555;
                    border-radius: 4px;
                    min-height: 10px;
                    color: white;
                }
            """)
            self.table.setCellWidget(row_count, col_idx, combo)
            
        # Highlight required columns in new row
        for col_idx, col_name in enumerate(self.columns):
            if col_name in self.required_columns and col_idx not in self.dropdown_columns:
                item = QTableWidgetItem("")
                item.setBackground(QColor(70, 130, 180, 50))
                self.table.setItem(row_count, col_idx, item)
                
        # Scroll to new row
        self.table.scrollToBottom()
        
    def remove_row(self):
        """Remove selected rows with confirmation."""
        selected_rows = set(index.row() for index in self.table.selectionModel().selectedRows())
        
        if not selected_rows:
            QMessageBox.information(self, "No Selection", "Please select one or more rows to delete.")
            return
            
        reply = QMessageBox.question(
            self, "Confirm Deletion",
            f"Are you sure you want to delete {len(selected_rows)} row(s)?",
            QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No,
            QMessageBox.StandardButton.No
        )
        
        if reply == QMessageBox.StandardButton.Yes:
            # Store data for undo functionality
            for row in sorted(selected_rows, reverse=True):
                row_data = []
                for col in range(self.table.columnCount()):
                    widget = self.table.cellWidget(row, col)
                    if isinstance(widget, QComboBox):
                        row_data.append(widget.currentText())
                    else:
                        item = self.table.item(row, col)
                        row_data.append(item.text() if item else "")
                        
                self.undo_stack.append((row, row_data))
                self.table.removeRow(row)
                
    def undo_remove(self):
        """Undo the last row removal."""
        if not self.undo_stack:
            QMessageBox.information(self, "Nothing to Undo", "No recent deletions to undo.")
            return
            
        row_idx, row_data = self.undo_stack.pop()
        self.table.insertRow(row_idx)
        
        for col_idx, data in enumerate(row_data):
            if col_idx in self.dropdown_columns:
                combo = QComboBox()
                combo.addItems(self.dropdown_columns[col_idx])
                combo.setCurrentText(data)
                combo.setStyleSheet("""
                    QComboBox {
                        background-color: #3a3a3a;
                        border-radius: 4px;
                        min-height: 10px;
                        color: white;
                    }
                """)
                self.table.setCellWidget(row_idx, col_idx, combo)
            else:
                item = QTableWidgetItem(data)
                if col_idx in [i for i, col in enumerate(self.columns) if col in self.required_columns]:
                    item.setBackground(QColor(70, 130, 180, 50))
                self.table.setItem(row_idx, col_idx, item)
                
    def import_data(self):
        """Import data from CSV file."""
        file_path, _ = QFileDialog.getOpenFileName(
            self, f"Import {self.tab_name} Data", "",
            "CSV Files (*.csv);;All Files (*)"
        )
        
        if file_path:
            try:
                import shutil
                shutil.copy(file_path, self.csv_file)
                self.load_data()
                QMessageBox.information(self, "Success", f"Data imported successfully from {os.path.basename(file_path)}")
            except Exception as e:
                QMessageBox.critical(self, "Import Error", f"Failed to import data:\n{str(e)}")
                
    def export_data(self):
        """Export current table data to CSV file."""
        file_path, _ = QFileDialog.getSaveFileName(
            self, f"Export {self.tab_name} Data", f"{self.tab_name}_export.csv",
            "CSV Files (*.csv);;All Files (*)"
        )
        
        if file_path:
            try:
                self.save_data()
                import shutil
                shutil.copy(self.csv_file, file_path)
                QMessageBox.information(self, "Success", f"Data exported successfully to {os.path.basename(file_path)}")
            except Exception as e:
                QMessageBox.critical(self, "Export Error", f"Failed to export data:\n{str(e)}")
                
    def validate_data(self) -> Tuple[bool, str]:
        """Validate current table data."""
        errors = []
        
        for row in range(self.table.rowCount()):
            for col, col_name in enumerate(self.columns):
                if col_name in self.required_columns:
                    widget = self.table.cellWidget(row, col)
                    if isinstance(widget, QComboBox):
                        value = widget.currentText()
                    else:
                        item = self.table.item(row, col)
                        value = item.text() if item else ""
                        
                    if not value.strip():
                        errors.append(f"Row {row + 1}: {col_name} is required")
                        
        return len(errors) == 0, "\n".join(errors)
        
    def save_data(self) -> bool:
        """Save current table data to CSV file."""
        is_valid, error_msg = self.validate_data()
        if not is_valid:
            QMessageBox.warning(self, "Validation Error", f"Please fix the following errors:\n{error_msg}")
            return False
            
        try:
            headers = self.columns
            rows = []
            
            for row in range(self.table.rowCount()):
                row_data = []
                for col in range(self.table.columnCount()):
                    widget = self.table.cellWidget(row, col)
                    if isinstance(widget, QComboBox):
                        row_data.append(widget.currentText())
                    else:
                        item = self.table.item(row, col)
                        row_data.append(item.text() if item else "")
                rows.append(row_data)
                
            os.makedirs(DATA_DIR, exist_ok=True)
            with open(self.csv_file, 'w', newline='', encoding='utf-8') as f:
                writer = csv.writer(f)
                writer.writerow(headers)
                writer.writerows(rows)
                
            return True
            
        except Exception as e:
            QMessageBox.critical(self, "Save Error", f"Failed to save data:\n{str(e)}")
            return False

    def set_config_from_dict(self, config: Dict[str, Any]):
        """Set configuration from dictionary."""
        try:
            if "daily_start_time" in config:
                time_parts = config["daily_start_time"].split(":")
                self.daily_start_time.setTime(QTime(int(time_parts[0]), int(time_parts[1])))
                
            if "daily_end_time" in config:
                time_parts = config["daily_end_time"].split(":")
                self.daily_end_time.setTime(QTime(int(time_parts[0]), int(time_parts[1])))
                
            self.class_duration.setValue(config.get("class_duration_minutes", 60))
            self.working_days.setText(config.get("working_days", "Mon,Tue,Wed,Thu,Fri"))
            self.max_classes_day.setValue(config.get("max_classes_per_day", 4))
            self.preferred_room_gap.setValue(config.get("preferred_room_gap", 1))
            self.allow_back_to_back.setChecked(config.get("allow_back_to_back", True))
            self.allow_mixed_sections.setChecked(config.get("allow_mixed_sections", True))
            self.allow_evening_classes.setChecked(config.get("allow_evening_classes", True))
            self.population_size.setValue(config.get("population_size", 30))
            self.generations.setValue(config.get("generations", 100))
            self.generate_visual.setChecked(config.get("generate_visual_timetables", True))
            
        except Exception as e:
            QMessageBox.warning(self, "Config Error", f"Error setting configuration: {str(e)}")
            
    def save_config(self):
        """Save current configuration to file."""
        try:
            os.makedirs(DATA_DIR, exist_ok=True)
            config = self.get_config_dict()
            with open(self.config_file, 'w') as f:
                json.dump(config, f, indent=4)
            return True
        except Exception as e:
            QMessageBox.critical(self, "Save Error", f"Failed to save configuration:\n{str(e)}")
            return False
            
    def load_config(self):
        """Load configuration from file."""
        if os.path.exists(self.config_file):
            try:
                with open(self.config_file, 'r') as f:
                    config = json.load(f)
                self.set_config_from_dict(config)
            except Exception as e:
                QMessageBox.warning(self, "Load Error", f"Failed to load configuration:\n{str(e)}")
                self.reset_to_defaults()
        else:
            self.reset_to_defaults()
            
    def reset_to_defaults(self):
        """Reset all configuration to default values."""
        default_config = {
            "working_days": "Mon,Tue,Wed,Thu,Fri",
            "daily_start_time": "08:00",
            "daily_end_time": "15:30",
            "class_duration_minutes": 60,
            "max_classes_per_day": 4,
            "preferred_room_gap": 1,
            "allow_back_to_back": True,
            "allow_mixed_sections": True,
            "allow_evening_classes": True,
            "population_size": 30,
            "generations": 100,
            "generate_visual_timetables": True
        }
        self.set_config_from_dict(default_config)
        
    def validate_config(self) -> Tuple[bool, str]:
        """Validate current configuration."""
        errors = []
        
        # Time validation
        if self.daily_start_time.time() >= self.daily_end_time.time():
            errors.append("Daily start time must be before end time")
            
        # Working days validation
        working_days = self.working_days.text().strip()
        if not working_days:
            errors.append("Working days cannot be empty")
        else:
            valid_days = {'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'}
            days = [day.strip() for day in working_days.split(',')]
            invalid_days = [day for day in days if day not in valid_days]
            if invalid_days:
                errors.append(f"Invalid working days: {', '.join(invalid_days)}")
                
        return len(errors) == 0, "\n".join(errors)

    def run_optimization(self):
        """Run the scheduling optimization with enhanced UI feedback."""
        # First save all data
        self.save_all_data()
        
        # Validate before running
        validation_errors = []
        course_valid, course_error = self.course_tab.validate_data()
        room_valid, room_error = self.room_tab.validate_data()
        config_valid, config_error = self.config_tab.validate_config()
        
        if not course_valid:
            validation_errors.append(f"Courses: {course_error}")
        if not room_valid:
            validation_errors.append(f"Rooms: {room_error}")
        if not config_valid:
            validation_errors.append(f"Configuration: {config_error}")
            
        if validation_errors:
            QMessageBox.critical(self, "Validation Failed", 
                               "Please fix the following errors before running:\n\n" + "\n".join(validation_errors))
            return
        
        # Define scripts to run (updated with correct names)
        scripts_to_run = [
            "enhanced_run_optimizer.py",        # Main optimizer script
            "generate_visual_timetables.py"     # Visual timetable generator
        ]
        
        # Check if first script exists (required)
        if not os.path.exists(scripts_to_run[0]):
            QMessageBox.critical(self, "Script Missing", 
                               f"Required optimizer script not found: {scripts_to_run[0]}")
            return
        
        # Remove second script if it doesn't exist (optional)
        if len(scripts_to_run) > 1 and not os.path.exists(scripts_to_run[1]):
            scripts_to_run = scripts_to_run[:1]
            self.update_console(f"Note: {scripts_to_run[1]} not found, skipping visual generation")
            
        # Start optimization
        self.start_optimization_process(scripts_to_run)
            
    def start_optimization_process(self, scripts_to_run):
        """Start the optimization process in background thread."""
        config_dict = self.config_tab.get_config_dict()
        
        # Update UI for running state
        self.run_btn.setEnabled(False)
        self.progress_bar.setVisible(True)
        self.console_output.setVisible(True)
        self.console_output.clear()
        self.progress_bar.setValue(0)
        
        # Start worker thread
        self.optimization_worker = OptimizationWorker(config_dict, scripts_to_run)
        self.optimization_worker.progress_update.connect(self.update_console)
        self.optimization_worker.progress_value.connect(self.progress_bar.setValue)
        self.optimization_worker.finished_signal.connect(self.optimization_finished)
        self.optimization_worker.start()
        
    def update_console(self, message):
        """Update console output and status."""
        self.console_output.append(message)
        self.status_label.setText(message)
        
        # Auto-scroll to bottom
        cursor = self.console_output.textCursor()
        cursor.movePosition(cursor.MoveOperation.End)
        self.console_output.setTextCursor(cursor)
        
    def optimization_finished(self, success, message):
        """Handle optimization completion."""
        self.run_btn.setEnabled(True)
        
        if success:
            self.status_label.setText("✅ Optimization completed successfully!")
            self.progress_bar.setValue(100)
            QMessageBox.information(self, "Success", message)
            
            # Check if output files were created
            output_locations = []
            if os.path.exists("output"):
                output_locations.append("output/ directory")
            if os.path.exists("exports"):
                output_locations.append("exports/ directory")
                
            if output_locations:
                self.statusbar.showMessage(f"Results saved to: {', '.join(output_locations)}", 5000)
        else:
            self.status_label.setText("❌ Optimization failed")
            self.progress_bar.setVisible(False)
            QMessageBox.critical(self, "Optimization Failed", message)
            
        # Keep progress bar and console visible for a moment if successful
        if success:
            QTimer.singleShot(3000, lambda: self.progress_bar.setVisible(False))
            QTimer.singleShot(3000, lambda: self.console_output.setVisible(False))
        else:
            self.console_output.setVisible(False)
