int binarysearch_title(Book arr [], int size, string input)
{
	int first = 0, // First array element 
		last = size - 1, // Last array element 
		middle, // Midpoint of search 
		position = -1;// Position of search value 
	bool found = false; // Flag 
	while (!found && first <= last)
	{
		middle = (first + last) / 2; // Calculate midpoint 
		if (arr[middle].title == input) // If value is found at mid 
		{
			found = true;
			position = middle;
		}
		else if (arr[middle].title > input) // If value is in lower half 
			last = middle - 1;
		else
			first = middle + 1; // If value is in upper half 
	}
	return position;
}

int linearsearch_title(Book arr[], int size, string input)
{
	int index = 0; // Used as a subscript to search array
	int position = -1; // To record position of search value
	bool found = false; // Flag to indicate if the value was found
	while (index < size && !found)
	{
		if (arr[index].title == input)
		{
			found = true;
			position = index;
		}
		index++;
	}
	return position;

}

int linearsearch_price(Book arr[], int size, double input)
{
	int index = 0; // Used as a subscript to search array
	int position = -1; // To record position of search value
	bool found = false; // Flag to indicate if the value was found
	while (index < size && !found)
	{
		if (arr[index].price == input)
		{
			found = true;
			position = index;
		}
		index++;
	}
	return position;

}
