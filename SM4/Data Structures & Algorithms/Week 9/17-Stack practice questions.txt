struct Node {
    int data;
    Node* next;
};

class Stack {
    Node* top;

public:
    Stack() { top = nullptr; }

    void push(int x) {
        Node* temp = new Node;
        temp->data = x;
        temp->next = top;
        top = temp;
    }

    void pop() {
        if (top != nullptr) {
            Node* temp = top;
            top = top->next;
            delete temp;
        }
    }

    int peek() {
        if (top != nullptr)
            return top->data;
        return -1; // error value
    }

    bool isEmpty() {
        return top == nullptr;
    }

int count() {
    int cnt = 0;
    Node* temp = top;
    while (temp != nullptr) {
        cnt++;
        temp = temp->next;
    }
    return cnt;
}
};

//Q1
int main() {
    Stack s;
    s.push(5);
    s.push(10);
    s.pop();
    s.push(15);
    cout << s.peek() << endl;
}
