#include <iostream>
using namespace std;

class CQueue
{
public:
    int* queueArray; // Pointer to array implemented as Queue 
    int queueSize; // Total size of the Queue
    int front;
    int rear;

    CQueue(int size);
    ~CQueue();
    bool isFull();
    bool isEmpty();
    bool enqueue(int num);
    bool dequeue(int& num);
    void makeNull();
    void print();
};

CQueue::CQueue(int size)
{
    front = rear = -1;
    queueArray = new int[size];
    queueSize = size;
}

CQueue::~CQueue() //destructor
{
    delete[] queueArray;
}

void CQueue::makeNull()
{
    front = -1;
    rear = -1;
}

bool CQueue::isEmpty()
{
    if (front == -1)
        return true; // we can check “rear” too
    else
        return false;
}

bool CQueue::isFull()
{
    if (((rear + 1) % queueSize) == front)
        return true;
    else
        return false;
}

bool CQueue::enqueue(int num)
{
    if (isFull()) {
        cout << "Overflow\n";
        return false;
    }
    if (isEmpty())
        rear = front = 0;
    else
        rear = (rear + 1) % queueSize;
    queueArray[rear] = num;
    return true;
}

bool CQueue::dequeue(int& num)
{
    if (isEmpty()) {
        cout << "Underflow\n";
        return false;
    }
    num = queueArray[front];
    if (front == rear) //only one element in the queue
        front = rear = -1;
    else
        front = (front + 1) % queueSize;
    return true;
}

void CQueue::print()
{
    if (isEmpty())
    {
        cout << "Queue is empty.\n";
        return;
    }

    int i = front;
    while (true)
    {
        cout << queueArray[i] << " ";
        if (i == rear)
            break;
        i = (i + 1) % queueSize;
    }

    cout << endl << endl;
}
