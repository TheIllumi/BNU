#include <iostream>
using namespace std;

class IntQueue
{
public:
    int* queueArray; // Pointer to array implemented as Queue 
    int queueSize; // Total size of the Queue
    int front;
    int rear;
    int numItems; // Number of items currently in the Queue

    IntQueue(int);
    ~IntQueue();
    bool isEmpty();
    bool isFull();
    bool enqueue(int);
    bool dequeue(int&);
    void makeNull();
    void print();
};

IntQueue::IntQueue(int input)
{
    queueArray = new int[input]; // Pointer to array implemented as Queue 
    queueSize = input; // Total size of the Queue
    front = -1;
    rear = -1;
    numItems = 0; // Number of items currently in the Queue
}

IntQueue::~IntQueue()
{
    delete[] queueArray;
}

bool IntQueue::isEmpty()
{
    if (numItems == 0)
        return true;
    else
        return false;
}

bool IntQueue::isFull()
{
    if (numItems == queueSize)
        return true;
    else
        return false;
}

bool IntQueue::enqueue(int num)
{
    if (isFull())
    {
        cout << "Overflow.\n";
        return false;
    }
    // Calculate the new rear position
    rear = (rear + 1) % queueSize;
    // Insert new item
    queueArray[rear] = num;
    // Update item count
    numItems++;
    return true;
}

bool IntQueue::dequeue(int& num)
{
    if (isEmpty())
    {
        cout << "The queue is empty.\n";
        return false;
    }
    // Move front
    front = (front + 1) % queueSize;
    // Retrieve the front item
    num = queueArray[front];
    // Update item count
    numItems--;
    return true;
}

void IntQueue::makeNull()
{
    front = -1;
    rear = -1;
    numItems = 0;
}

//can use numItems as well to iterate over the Queue
void IntQueue::print()
{
    if (isEmpty()) 
    {
        cout << "Queue is empty.\n";
        return;
    }

    int i = front + 1;
    while (true) 
    {
        cout << queueArray[i] << " ";
        if (i == rear)
            break;
        i = (i + 1) % queueSize;
    }

    cout << endl << endl;
}
