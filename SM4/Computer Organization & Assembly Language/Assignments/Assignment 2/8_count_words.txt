.model small
.stack 100h
.data
    str db 'hello world$', 0
.code
main:
    mov ax, @data
    mov ds, ax

    lea si, str
    call COUNT_WORDS
    ; Result in AL

    mov ah, 4ch
    int 21h

COUNT_WORDS proc
    xor al, al
    mov bl, 0
next_char:
    lodsb
    cmp al, 0
    je finish
    cmp al, ' '
    je space
    cmp bl, 1
    je next_char
    inc al
    mov bl, 1
    jmp next_char
space:
    mov bl, 0
    jmp next_char
finish:
    ret
COUNT_WORDS endp
end main
